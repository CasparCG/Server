# Assert CEF provider selection without requiring a compiler or real CEF libraries.
cmake_minimum_required(VERSION 3.28)
project(CEFSelectionTest NONE)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}" "${CMAKE_CURRENT_LIST_DIR}/../../../../src/CMakeModules")
include(SelectLinuxCEF)

if(EXPECTED_CEF_SOURCE STREQUAL "download")
    if(USE_SYSTEM_CEF OR TARGET CEF::CEF)
        message(FATAL_ERROR "Expected downloaded CEF")
    endif()
elseif(EXPECTED_CEF_SOURCE STREQUAL "ppa")
    get_target_property(libraries CEF::CEF INTERFACE_LINK_LIBRARIES)
    if(NOT USE_SYSTEM_CEF OR NOT libraries STREQUAL "CasparCG::CEF")
        message(FATAL_ERROR "Expected CasparCG package to take priority")
    endif()
elseif(EXPECTED_CEF_SOURCE STREQUAL "system")
    if(NOT USE_SYSTEM_CEF OR NOT TARGET CEF::Library OR NOT CEF_API_VERSION EQUAL 15000)
        message(FATAL_ERROR "Expected distro CEF with API 15000")
    endif()
else()
    message(FATAL_ERROR "Missing expected CEF source")
endif()
